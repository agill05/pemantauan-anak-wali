const SPREADSHEET_ID = SpreadsheetApp.getActiveSpreadsheet().getId();
const TOKEN_LIFETIME_MS = 15 * 60 * 1000;
const PASSWORD_PEPPER = "AnakWali-SMPN1TalagaJaya-2026";
const TIMEZONE_WITA = "Asia/Makassar";

function doPost(e) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);
  } catch (err) {
    return responseJSON({ status: "error", message: "Server sedang sibuk. Silakan coba beberapa saat lagi." });
  }

  try {
    if (!e || !e.postData || !e.postData.contents) {
      return responseJSON({ status: "error", message: "Payload request kosong." });
    }

    const contents = JSON.parse(e.postData.contents);
    const action = contents.action;
    const token = contents.token || null;
    const payload = contents.payload || {};

    if (action === "login") return responseJSON(handleLogin(payload));
    if (action === "getMagicLinkData") return responseJSON(handleGetMagicLinkData(payload));

    const authUser = validateToken(token);
    if (!authUser) {
      return responseJSON({ status: "error", message: "Sesi telah berakhir atau token tidak valid. Silakan login kembali." });
    }

    if (action === "validateSession") return responseJSON({ status: "success", user: authUser });
    if (action === "refreshToken") return responseJSON(handleRefreshToken(authUser));
    if (action === "changePassword") return responseJSON(handleChangePassword(authUser, payload));

    let responseData = null;

    switch (action) {
      case "getBootstrapData": responseData = handleGetBootstrapData(authUser); break;
      case "getDashboardData": responseData = handleGetDashboardData(authUser); break;
      case "saveGuru": responseData = handleSaveGuru(authUser, payload); break;
      case "deleteGuru": responseData = handleDeleteGuru(authUser, payload); break;
      case "saveSiswa": responseData = handleSaveSiswa(authUser, payload); break;
      case "deleteSiswa": responseData = handleDeleteSiswa(authUser, payload); break;
      case "saveKelas": responseData = handleSaveKelas(authUser, payload); break;
      case "deleteKelas": responseData = handleDeleteKelas(authUser, payload); break;
      case "getAbsensi": responseData = handleGetAbsensi(authUser, payload); break;
      case "saveAbsensi": responseData = handleSaveAbsensi(authUser, payload); break;
      case "getKebiasaan": responseData = handleGetKebiasaan(authUser, payload); break;
      case "saveKebiasaan": responseData = handleSaveKebiasaan(authUser, payload); break;
      case "getKeagamaan": responseData = handleGetKeagamaan(authUser, payload); break;
      case "saveKeagamaan": responseData = handleSaveKeagamaan(authUser, payload); break;
      case "deleteKeagamaan": responseData = handleDeleteKeagamaan(authUser, payload); break;
      case "getAkademik": responseData = handleGetAkademik(authUser, payload); break;
      case "saveAkademik": responseData = handleSaveAkademik(authUser, payload); break;
      case "deleteAkademik": responseData = handleDeleteAkademik(authUser, payload); break;
      case "getPrestasi": responseData = handleGetPrestasi(authUser, payload); break;
      case "savePrestasi": responseData = handleSavePrestasi(authUser, payload); break;
      case "deletePrestasi": responseData = handleDeletePrestasi(authUser, payload); break;
      case "getPembinaan": responseData = handleGetPembinaan(authUser, payload); break;
      case "savePembinaan": responseData = handleSavePembinaan(authUser, payload); break;
      case "deletePembinaan": responseData = handleDeletePembinaan(authUser, payload); break;
      case "getDetailSiswa": responseData = handleGetDetailSiswa(authUser, payload); break;
      case "getLaporanRekap": responseData = handleGetLaporanRekap(authUser, payload); break;
      case "generateMagicLink": responseData = handleGenerateMagicLink(authUser, payload); break;
      case "updateSelfProfile": responseData = handleUpdateSelfProfile(authUser, payload); break; // <-- Penanganan Action Baru
      default: responseData = { status: "error", message: "Aksi tidak dikenal: " + action };
    }

    return responseJSON(responseData);

  } catch (err) {
    return responseJSON({ status: "error", message: err.toString() });
  } finally {
    lock.releaseLock();
  }
}

function responseJSON(data) {
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

function getSheet(sheetName) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  let sheet = ss.getSheetByName(sheetName);
  if (!sheet) sheet = ss.insertSheet(sheetName);
  return sheet;
}

function getSheetDataCached(sheetName) {
  const cache = CacheService.getScriptCache();
  const cacheKey = "cache_sheet_" + sheetName;
  const cached = cache.get(cacheKey);

  if (cached) {
    try {
      return JSON.parse(cached);
    } catch (e) {}
  }

  const data = getSheetData(sheetName);
  try {
    const jsonStr = JSON.stringify(data);
    if (jsonStr.length < 95000) {
      cache.put(cacheKey, jsonStr, 600);
    }
  } catch (e) {}

  return data;
}

function clearSheetCache(sheetName) {
  try {
    const cache = CacheService.getScriptCache();
    if (sheetName) {
      cache.remove("cache_sheet_" + sheetName);
    } else {
      const keys = ["Siswa", "Guru", "Kelas", "Admin", "Kehadiran", "Kebiasaan", "Hafalan", "Akademik", "Prestasi", "Pembinaan"];
      keys.forEach(k => cache.remove("cache_sheet_" + k));
    }
  } catch (e) {}
}

function sanitizeForSheet(value) {
  if (value === null || value === undefined) return "";
  if (typeof value === "string" && /^[=+\-@\t\r]/.test(value)) return "'" + value;
  return value;
}

function formatDateWITA(val) {
  if (val === null || val === undefined || val === "") return "";
  if (val instanceof Date) {
    if (val.getFullYear() < 1920) return "";
    return Utilities.formatDate(val, TIMEZONE_WITA, "yyyy-MM-dd");
  }
  let str = String(val).trim();
  if (str.indexOf("T") !== -1) str = str.split("T")[0];
  if (/^\d{4}-\d{2}-\d{2}$/.test(str)) return str;
  try {
    const d = new Date(str);
    if (!isNaN(d.getTime()) && d.getFullYear() >= 1920) {
      return Utilities.formatDate(d, TIMEZONE_WITA, "yyyy-MM-dd");
    }
  } catch (e) {}
  return str;
}

function formatTimeWITA(val) {
  if (!val) return Utilities.formatDate(new Date(), TIMEZONE_WITA, "HH:mm");
  if (val instanceof Date) return Utilities.formatDate(val, TIMEZONE_WITA, "HH:mm");
  const str = String(val).trim();
  if (str.indexOf("T") !== -1) {
    try {
      return Utilities.formatDate(new Date(str), TIMEZONE_WITA, "HH:mm");
    } catch (e) {
      return str;
    }
  }
  return str;
}

function getSheetData(sheetName) {
  const sheet = getSheet(sheetName);
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  const headers = data[0].map(h => String(h).trim());
  const rows = [];
  for (let i = 1; i < data.length; i++) {
    let rowObj = {};
    let isEmptyRow = true;
    for (let j = 0; j < headers.length; j++) {
      let val = data[i][j];
      if (val !== "" && val !== null && val !== undefined) isEmptyRow = false;
      if (headers[j] === "tanggal" || headers[j] === "created_at" || headers[j] === "jadwal_pantau") {
        val = formatDateWITA(val);
      }
      rowObj[headers[j]] = val;
    }
    if (!isEmptyRow) rows.push(rowObj);
  }
  return rows;
}

function updateSheetRow(sheetName, id, updatedObj) {
  const sheet = getSheet(sheetName);
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return false;
  const headers = data[0].map(h => String(h).trim());
  const idColIndex = headers.indexOf("id");

  if (idColIndex === -1) return false;

  for (let i = 1; i < data.length; i++) {
    if (String(data[i][idColIndex]) === String(id)) {
      for (let j = 0; j < headers.length; j++) {
        const key = headers[j];
        if (updatedObj[key] !== undefined) {
          let valToSave = updatedObj[key];
          if (key === "tanggal" || key === "jadwal_pantau") valToSave = formatDateWITA(valToSave);
          sheet.getRange(i + 1, j + 1).setValue(sanitizeForSheet(valToSave));
        }
      }
      clearSheetCache(sheetName);
      return true;
    }
  }
  return false;
}

function deleteRowById(sheetName, id) {
  const sheet = getSheet(sheetName);
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return { status: "error", message: "Data kosong." };

  const headers = data[0].map(h => String(h).trim());
  const idColIndex = headers.indexOf("id");
  if (idColIndex === -1) return { status: "error", message: "Kolom ID tidak ditemukan." };

  for (let i = 1; i < data.length; i++) {
    if (String(data[i][idColIndex]) === String(id)) {
      sheet.deleteRow(i + 1);
      clearSheetCache(sheetName);
      return { status: "success", message: "Data berhasil dihapus." };
    }
  }
  return { status: "error", message: "ID tidak ditemukan." };
}

function logActivity(userId, role, action, details) {
  try {
    const sheet = getSheet("Logs");
    const timestamp = Utilities.formatDate(new Date(), TIMEZONE_WITA, "yyyy-MM-dd HH:mm:ss");
    sheet.appendRow([Date.now(), timestamp, userId, role, action, JSON.stringify(details)]);
  } catch (e) {}
}

function getSecretKey() {
  const props = PropertiesService.getScriptProperties();
  let key = props.getProperty("APP_SECRET_KEY");
  if (!key) {
    key = Utilities.getUuid();
    props.setProperty("APP_SECRET_KEY", key);
  }
  return key;
}

function hashPassword(plainPassword) {
  const bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, String(plainPassword) + PASSWORD_PEPPER);
  return bytes.map(b => (b < 0 ? b + 256 : b).toString(16).padStart(2, '0')).join("");
}

function getDefaultPasswordForRole(role) {
  return role === "admin" ? "admin123" : (role === "guru" ? "guru123" : "siswa123");
}

function generateToken(userId, role) {
  const expiry = Date.now() + TOKEN_LIFETIME_MS;
  const payload = String(userId) + "|" + role + "|" + expiry;
  const encodedPayload = Utilities.base64EncodeWebSafe(payload);
  const signature = Utilities.base64EncodeWebSafe(Utilities.computeHmacSha256Signature(payload, getSecretKey()));
  return encodedPayload + "." + signature;
}

function validateToken(token) {
  if (!token || typeof token !== "string" || token.indexOf(".") === -1) return null;
  try {
    const parts = token.split(".");
    const payload = Utilities.newBlob(Utilities.base64DecodeWebSafe(parts[0])).getDataAsString();
    const expectedSig = Utilities.base64EncodeWebSafe(Utilities.computeHmacSha256Signature(payload, getSecretKey()));
    if (parts[1] !== expectedSig) return null;

    const p = payload.split("|");
    if (Number(p[2]) < Date.now()) return null;

    const sheetName = p[1] === "admin" ? "Admin" : (p[1] === "guru" ? "Guru" : "Siswa");
    const rawUser = getSheetDataCached(sheetName).find(r => String(r.id) === String(p[0]));
    if (!rawUser) return null;

    const user = JSON.parse(JSON.stringify(rawUser));
    const storedHash = user.password;
    delete user.password;
    user.role = p[1];
    user.mustChangePassword = (storedHash === hashPassword(getDefaultPasswordForRole(p[1])));

    return user;
  } catch (e) {
    return null;
  }
}

function handleLogin(payload) {
  const { role, username, password } = payload;
  if (!role || !username || !password) {
    return { status: "error", message: "Role, Username, dan Kata Sandi wajib diisi." };
  }

  let sheetName = role === "admin" ? "Admin" : (role === "guru" ? "Guru" : "Siswa");
  const users = getSheetDataCached(sheetName);
  const user = users.find(u => String(u.username).trim().toLowerCase() === String(username).trim().toLowerCase());

  const errMessage = "Username atau kata sandi yang Anda masukkan salah.";
  if (!user) return { status: "error", message: errMessage };

  const inputHash = hashPassword(password);
  const isMatch = (user.password === inputHash) || (String(user.password) === String(password));

  if (!isMatch) return { status: "error", message: errMessage };

  if (user.password !== inputHash) {
    updateSheetRow(sheetName, user.id, { password: inputHash });
  }

  const token = generateToken(user.id, role);
  const mustChangePassword = (inputHash === hashPassword(getDefaultPasswordForRole(role)));
  delete user.password;
  user.role = role;
  user.mustChangePassword = mustChangePassword;

  logActivity(user.id, role, "LOGIN", { username: user.username });
  return { status: "success", token, user, mustChangePassword };
}

function handleRefreshToken(authUser) {
  const newToken = generateToken(authUser.id, authUser.role);
  return { status: "success", token: newToken, user: authUser };
}

function handleChangePassword(authUser, payload) {
  const { oldPassword, newPassword } = payload;
  if (!oldPassword || !newPassword) return { status: "error", message: "Password lama dan baru wajib diisi." };

  const sheetName = authUser.role === "admin" ? "Admin" : (authUser.role === "guru" ? "Guru" : "Siswa");
  const users = getSheetDataCached(sheetName);
  const targetUser = users.find(u => String(u.id) === String(authUser.id));

  if (!targetUser || targetUser.password !== hashPassword(oldPassword)) {
    return { status: "error", message: "Password lama Anda tidak sesuai." };
  }

  updateSheetRow(sheetName, authUser.id, { password: hashPassword(newPassword) });
  logActivity(authUser.id, authUser.role, "CHANGE_PASSWORD", { user_id: authUser.id });
  return { status: "success", message: "Password berhasil diperbarui." };
}

function handleGetBootstrapData(authUser) {
  const initial = handleGetInitialData(authUser);
  const dashboard = handleGetDashboardData(authUser);
  return {
    status: "success",
    data: { initial: initial.data, dashboard: dashboard.data }
  };
}

function handleGetInitialData(authUser) {
  const role = authUser.role;
  const kelasList = getSheetDataCached("Kelas");
  const guruList = getSheetDataCached("Guru").map(g => { delete g.password; return g; });
  const siswaList = getSheetDataCached("Siswa").map(s => { delete s.password; return s; });

  let allowedSiswa = [];
  if (role === "admin") {
    allowedSiswa = siswaList;
  } else if (role === "guru") {
    const kls = kelasList.find(k => String(k.guru_id) === String(authUser.id));
    allowedSiswa = kls ? siswaList.filter(s => String(s.kelas_id) === String(kls.id)) : [];
  } else if (role === "siswa") {
    allowedSiswa = siswaList.filter(s => String(s.id) === String(authUser.id));
  }

  return {
    status: "success",
    data: { kelas: kelasList, guru: guruList, siswa: allowedSiswa }
  };
}

function handleGetDashboardData(authUser) {
  const role = authUser.role;
  const today = formatDateWITA(new Date());

  const allSiswa = getSheetDataCached("Siswa").map(s => { delete s.password; return s; });
  const allKelas = getSheetDataCached("Kelas");
  const allAbsensi = getSheetDataCached("Kehadiran");
  const allAkademik = getSheetDataCached("Akademik");
  const allPembinaan = getSheetDataCached("Pembinaan");

  let mySiswaList = [];
  if (role === "admin") {
    mySiswaList = allSiswa;
  } else if (role === "guru") {
    const kls = allKelas.find(k => String(k.guru_id) === String(authUser.id));
    mySiswaList = kls ? allSiswa.filter(s => String(s.kelas_id) === String(kls.id)) : [];
  } else if (role === "siswa") {
    mySiswaList = allSiswa.filter(s => String(s.id) === String(authUser.id));
  }

  const mySiswaIds = mySiswaList.map(s => String(s.id));
  const todayAbsensi = allAbsensi.filter(a => formatDateWITA(a.tanggal) === today && mySiswaIds.includes(String(a.siswa_id)));
  const totalHadirToday = todayAbsensi.filter(a => a.status === "H" || a.status === "T").length;

  const priorityList = [];
  const activePembinaanStatuses = ["Pemantauan", "Dalam pembinaan", "Perlu tindak lanjut"];

  mySiswaList.forEach(siswa => {
    const sId = String(siswa.id);
    const flags = [];

    const totalAlpa = allAbsensi.filter(a => String(a.siswa_id) === sId && a.status === "A").length;
    if (totalAlpa >= 3) flags.push({ jenis: "absensi", pesan: `Banyak alpa (${totalAlpa} kali)` });

    const lowGrades = allAkademik.filter(a => String(a.siswa_id) === sId && Number(a.nilai_akhir || 0) < Number(a.kktp || 75));
    if (lowGrades.length >= 2) flags.push({ jenis: "akademik", pesan: `Nilai akademik kurang pada ${lowGrades.length} mapel` });

    const activePem = allPembinaan.filter(p => String(p.siswa_id) === sId && activePembinaanStatuses.includes(String(p.status)));
    if (activePem.length > 0) flags.push({ jenis: "pembinaan", pesan: `Ada ${activePem.length} catatan pembinaan aktif` });

    if (flags.length > 0) priorityList.push({ siswa, indicators: flags });
  });

  const agendaList = allPembinaan
    .filter(p => mySiswaIds.includes(String(p.siswa_id)) && p.jadwal_pantau && formatDateWITA(p.jadwal_pantau) >= today)
    .sort((a, b) => new Date(formatDateWITA(a.jadwal_pantau)) - new Date(formatDateWITA(b.jadwal_pantau)))
    .slice(0, 5)
    .map(p => {
      const s = mySiswaList.find(x => String(x.id) === String(p.siswa_id));
      return {
        id: p.id,
        siswa_id: p.siswa_id,
        nama_siswa: s ? s.nama : "Siswa",
        jadwal_pantau: formatDateWITA(p.jadwal_pantau),
        jenis: p.jenis,
        permasalahan: p.permasalahan,
        status: p.status
      };
    });

  return {
    status: "success",
    data: {
      total_siswa: mySiswaList.length,
      hadir_today: totalHadirToday,
      priority_list: priorityList,
      agenda_list: agendaList
    }
  };
}

function handleSaveGuru(authUser, payload) {
  if (authUser.role !== "admin") return { status: "error", message: "Akses ditolak." };
  const { id, username, password, nama, nip, no_hp, foto } = payload;
  if (!username || !nama) return { status: "error", message: "Username dan Nama Guru wajib diisi." };

  const sheet = getSheet("Guru");
  const data = getSheetDataCached("Guru");

  if (id) {
    const updatedObj = { username, nama, nip, no_hp, foto };
    if (password && password.trim() !== "") updatedObj.password = hashPassword(password);
    updateSheetRow("Guru", id, updatedObj);
    return { status: "success", message: "Data Guru berhasil diperbarui." };
  } else {
    const existing = data.find(g => String(g.username).toLowerCase() === String(username).toLowerCase());
    if (existing) return { status: "error", message: "Username sudah digunakan." };

    const newId = "GRU-" + Date.now();
    const usedDefaultPassword = !password || password.trim() === "";
    const finalPwd = usedDefaultPassword ? hashPassword(getDefaultPasswordForRole("guru")) : hashPassword(password);
    sheet.appendRow([newId, username, finalPwd, nama, nip || "", no_hp || "", foto || ""]);
    clearSheetCache("Guru");
    return {
      status: "success",
      message: usedDefaultPassword ? `Guru baru ditambahkan. Password diset ke "${getDefaultPasswordForRole("guru")}".` : "Guru baru berhasil ditambahkan.",
      usedDefaultPassword
    };
  }
}

function handleDeleteGuru(authUser, payload) {

  return deleteRowById("Guru", payload.id);
}

function handleSaveKelas(authUser, payload) {
  if (authUser.role !== "admin") return { status: "error", message: "Akses ditolak." };
  const { id, nama_kelas, guru_id } = payload;
  if (!nama_kelas) return { status: "error", message: "Nama kelas wajib diisi." };

  const sheet = getSheet("Kelas");
  if (id) {
    updateSheetRow("Kelas", id, { nama_kelas, guru_id });

  } else {
    const newId = "KLS-" + Date.now();
    sheet.appendRow([newId, nama_kelas, guru_id || ""]);
    clearSheetCache("Kelas");
    return { status: "success", message: "Kelas baru berhasil dibuat." };
  }
}

function handleDeleteKelas(authUser, payload) {

}

function handleGetAbsensi(authUser, payload) {
  const tanggal = formatDateWITA(payload.tanggal || new Date());
  const allAbsensi = getSheetDataCached("Kehadiran");
  
  const absensiMap = new Map();
  allAbsensi.forEach(a => {
    if (formatDateWITA(a.tanggal) === tanggal) {
      absensiMap.set(String(a.siswa_id), a);
    }
  });

  return { status: "success", data: Array.from(absensiMap.values()) };
}

function handleSaveAbsensi(authUser, payload) {
  if (authUser.role === "siswa") return { status: "error", message: "Siswa tidak dapat menyimpan presensi." };

  const targetTanggal = formatDateWITA(payload.tanggal || new Date());
  const items = payload.items || [];
  if (items.length === 0) return { status: "success", message: "Tidak ada data disave." };

  const sheet = getSheet("Kehadiran");
  const data = sheet.getDataRange().getValues();
  
  if (data.length === 0 || (data.length === 1 && data[0][0] === "")) {
    data[0] = ["id", "siswa_id", "tanggal", "status", "waktu_masuk"];
  }

  const headers = data[0].map(h => String(h).trim().toLowerCase());
  const idIdx = headers.indexOf("id") !== -1 ? headers.indexOf("id") : 0;
  const siswaIdIdx = headers.indexOf("siswa_id") !== -1 ? headers.indexOf("siswa_id") : 1;
  const tglIdx = headers.indexOf("tanggal") !== -1 ? headers.indexOf("tanggal") : 2;
  const statusIdx = headers.indexOf("status") !== -1 ? headers.indexOf("status") : 3;
  const waktuIdx = headers.indexOf("waktu_masuk") !== -1 ? headers.indexOf("waktu_masuk") : 4;

  const rowMap = new Map();
  for (let i = 1; i < data.length; i++) {
    const sId = String(data[i][siswaIdIdx]).trim();
    const tglStr = formatDateWITA(data[i][tglIdx]);
    if (sId && tglStr) {
      rowMap.set(sId + "_" + tglStr, i);
    }
  }

  const currentTime = formatTimeWITA(new Date());

  items.forEach(item => {
    const sId = String(item.siswa_id).trim();
    if (!sId) return;

    const waktu = item.waktu_masuk || currentTime;
    const statusVal = sanitizeForSheet(item.status);
    const key = sId + "_" + targetTanggal;

    if (rowMap.has(key)) {
      const idx = rowMap.get(key);
      data[idx][tglIdx] = targetTanggal;
      data[idx][statusIdx] = statusVal;
      data[idx][waktuIdx] = sanitizeForSheet(waktu);
    } else {
      const newId = "ABS-" + Date.now() + "-" + Math.floor(Math.random() * 1000);
      const newRow = new Array(headers.length).fill("");
      newRow[idIdx] = newId;
      newRow[siswaIdIdx] = sId;
      newRow[tglIdx] = targetTanggal;
      newRow[statusIdx] = statusVal;
      newRow[waktuIdx] = sanitizeForSheet(waktu);
      data.push(newRow);
      rowMap.set(key, data.length - 1);
    }
  });

  sheet.getRange(1, 1, data.length, headers.length).setValues(data);
  clearSheetCache("Kehadiran");
  return { status: "success", message: "Data presensi berhasil disimpan." };
}

      String(data[i][siswaIdIdx]) === String(siswa_id) &&

    };
  });

function generateMagicToken(siswaId) {
  const expiry = Date.now() + (15 * 60 * 1000);
  const payloadStr = String(siswaId) + "|" + expiry;
  const encodedPayload = Utilities.base64EncodeWebSafe(payloadStr);
  const signature = Utilities.base64EncodeWebSafe(Utilities.computeHmacSha256Signature(payloadStr, getSecretKey()));
  return encodedPayload + "." + signature;
}

function validateMagicToken(magicToken) {
  if (!magicToken || typeof magicToken !== "string" || magicToken.indexOf(".") === -1) return null;
  try {
    const parts = magicToken.split(".");
    const payloadStr = Utilities.newBlob(Utilities.base64DecodeWebSafe(parts[0])).getDataAsString();
    const expectedSig = Utilities.base64EncodeWebSafe(Utilities.computeHmacSha256Signature(payloadStr, getSecretKey()));
    
    if (parts[1] !== expectedSig) return null;

    const p = payloadStr.split("|");
    const siswaId = p[0];
    const expiry = Number(p[1]);

    if (Date.now() > expiry) {
      return { status: "expired", message: "Tautan Magic Link telah kedaluwarsa (berlaku 15 menit). Silakan minta tautan baru kepada Guru/Wali Kelas." };
    }

    return { status: "valid", siswaId };
  } catch (e) {
    return null;
  }
}

function handleGenerateMagicLink(authUser, payload) {

function handleUploadFotoProfil(payload, user) {
  try {
    if (!payload.fileData) {
      return { status: "error", message: "Data file tidak ditemukan." };
    }

    var role = (user.role || "siswa").toLowerCase();
    var folderId = FOLDER_IDS[role] || FOLDER_IDS["siswa"];

    var targetFolder = DriveApp.getFolderById(folderId);

    if (user.foto && user.foto.includes("googleusercontent.com/d/")) {
      deleteOldDriveFile(user.foto);
    }

    var contentType = payload.fileData.substring(5, payload.fileData.indexOf(';base64'));
    var bytes = Utilities.base64Decode(payload.fileData.substr(payload.fileData.indexOf('base64,') + 7));
    var fileName = user.role + "_" + user.id + "_" + Date.now() + ".jpg";
    var blob = Utilities.newBlob(bytes, contentType, fileName);

    var file = targetFolder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

    var photoUrl = "https://lh3.googleusercontent.com/d/" + file.getId();
    updateUserPhotoInSheet(user.role, user.id, photoUrl);

    return { 
      status: "success", 
      photoUrl: photoUrl,
      message: "Foto profil berhasil diperbarui!"

    return { status: "error", message: err.toString() };
  }
}

function getOrCreateFolder(folderName, parent) {
  var collection = parent ? parent.getFoldersByName(folderName) : DriveApp.getFoldersByName(folderName);
  if (collection.hasNext()) {
    return collection.next();
  } else {
    return parent ? parent.createFolder(folderName) : DriveApp.createFolder(folderName);
  }
}

function deleteOldDriveFile(oldUrl) {
  try {
    var fileId = oldUrl.replace("https://lh3.googleusercontent.com/d/", "");
    if (fileId) {
      DriveApp.getFileById(fileId).setTrashed(true);
    }
  } catch (e) {
    Logger.log("Gagal menghapus file lama: " + e.toString());
  }
}

function capitalizeFirst(string) {
  return string.charAt(0).toUpperCase() + string.slice(1).toLowerCase();
}

      if (payload.nama) updatedObj.nama = payload.nama;
      if (payload.username) updatedObj.username = payload.username;
      if (payload.no_hp !== undefined) updatedObj.no_hp = payload.no_hp;
    } else if (authUser.role === "guru") {
      if (payload.no_hp !== undefined) updatedObj.no_hp = payload.no_hp;
    } else if (authUser.role === "siswa") {
      if (payload.no_hp !== undefined) updatedObj.no_hp_ortu = payload.no_hp;
    }

    if (photoUrl) {
      updatedObj.foto = photoUrl;
    }

    const isUpdated = updateSheetRow(sheetName, authUser.id, updatedObj);

    if (isUpdated) {
      return {
        status: "success",
        photoUrl: photoUrl,
        message: "Profil berhasil diperbarui!"
      };
    } else {
      return { status: "error", message: "Gagal memperbarui profil di spreadsheet." };
    }
  } catch (err) {
    return { status: "error", message: err.toString() };
  }
}
