const SPREADSHEET_ID = SpreadsheetApp.getActiveSpreadsheet().getId();
const TOKEN_LIFETIME_MS = 15 * 60 * 1000; // 15 Menit
const PASSWORD_PEPPER = "AnakWali-SMPN1TalagaJaya-2026";
const TIMEZONE_WITA = "Asia/Makassar";

// Helper Concurrency Lock Engine untuk mencegah race condition
function withLock(callback) {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) {
    return { status: "error", message: "Sistem sedang sibuk memproses data lain. Silakan coba beberapa saat lagi." };
  }
  try {
    return callback();
  } finally {
    lock.releaseLock();
  }
}

// Helper Audit Logger untuk mencatat seluruh aktivitas sensitif
function logActivity(userId, role, action, details) {
  try {
    const sheet = getSheet("Logs");
    const timestamp = Utilities.formatDate(new Date(), TIMEZONE_WITA, "yyyy-MM-dd HH:mm:ss");
    sheet.appendRow([Date.now(), timestamp, userId, role, action, JSON.stringify(details)]);
  } catch (e) {
    // Fail-safe jika pencatatan log mengalami kendala
  }
}

// Helper Format Tanggal & Waktu WITA
function formatDateWITA(val) {
  if (!val) return "";
  if (val instanceof Date) return Utilities.formatDate(val, TIMEZONE_WITA, "yyyy-MM-dd");
  const str = String(val).trim();
  return str.indexOf("T") !== -1 ? str.split("T")[0] : str;
}

function formatTimeWITA(val) {
  if (!val) return Utilities.formatDate(new Date(), TIMEZONE_WITA, "HH:mm");
  if (val instanceof Date) return Utilities.formatDate(val, TIMEZONE_WITA, "HH:mm");
  const str = String(val).trim();
  if (str.indexOf("T") !== -1) {
    try {
      const d = new Date(str);
      return Utilities.formatDate(d, TIMEZONE_WITA, "HH:mm");
    } catch (e) {
      return str;
    }
  }
  return str;
}

// Router HTTP GET & POST
function doGet(e) {
  return responseJSON({ status: "success", message: "API Pemantauan Anak Wali Active (Optimized)", version: "3.0" });
}

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return responseJSON({ status: "error", message: "Payload request kosong." });
    }

    const requestData = JSON.parse(e.postData.contents);
    const action = requestData.action;
    const payload = requestData.payload || {};
    const token = requestData.token || null;

    // Rute Publik
    if (action === "login") return responseJSON(handleLogin(payload));
    if (action === "initDatabase") return responseJSON(initDatabaseSheets());

    // Validasi Sesi Terproteksi (Cache Accelerated)
    const authUser = validateToken(token);
    if (!authUser) {
      return responseJSON({ status: "session_expired", message: "Sesi Anda telah berakhir. Silakan login kembali." });
    }

    // Pemetaan Rute API Backend
    const routes = {
      validateSession: () => ({ status: "success", user: authUser }),
      getBootstrapData: () => handleGetBootstrapData(authUser),
      getInitialData: () => handleGetInitialData(authUser),
      getDashboardData: () => handleGetDashboardData(authUser),
      changePassword: () => withLock(() => handleChangePassword(authUser, payload)),
      refreshToken: () => handleRefreshToken(authUser),
      
      // Master Data & Operasi
      getAbsensi: () => handleGetAbsensi(authUser, payload),
      saveAbsensi: () => withLock(() => handleSaveAbsensi(authUser, payload)),
      saveGuru: () => withLock(() => handleSaveGuru(authUser, payload)),
      deleteGuru: () => withLock(() => handleDeleteGuru(authUser, payload)),
      saveSiswa: () => withLock(() => handleSaveSiswa(authUser, payload)),
      deleteSiswa: () => handleDeleteSiswa(authUser, payload),
      saveKelas: () => withLock(() => handleSaveKelas(authUser, payload)),
      deleteKelas: () => withLock(() => handleDeleteKelas(authUser, payload)),

      // Kebiasaan & Keagamaan
      getKebiasaan: () => handleGetKebiasaan(authUser, payload),
      saveKebiasaan: () => withLock(() => handleSaveKebiasaan(authUser, payload)),
      getKeagamaan: () => handleGetKeagamaan(authUser, payload),
      saveKeagamaan: () => withLock(() => handleSaveKeagamaan(authUser, payload)),
      deleteKeagamaan: () => withLock(() => handleDeleteKeagamaan(authUser, payload)),

      // Akademik, Prestasi, & Pembinaan
      getAkademik: () => handleGetAkademik(authUser, payload),
      saveAkademik: () => withLock(() => handleSaveAkademik(authUser, payload)),
      deleteAkademik: () => withLock(() => handleDeleteAkademik(authUser, payload)),
      getPrestasi: () => handleGetPrestasi(authUser, payload),
      savePrestasi: () => withLock(() => handleSavePrestasi(authUser, payload)),
      deletePrestasi: () => withLock(() => handleDeletePrestasi(authUser, payload)),
      getPembinaan: () => handleGetPembinaan(authUser, payload),
      savePembinaan: () => withLock(() => handleSavePembinaan(authUser, payload)),
      deletePembinaan: () => withLock(() => handleDeletePembinaan(authUser, payload)),

      // Detail & Rekap Laporan
      getDetailSiswa: () => handleGetDetailSiswa(authUser, payload),
      getLaporanRekap: () => handleGetLaporanRekap(authUser, payload)
    };

    const handler = routes[action];
    return responseJSON(handler ? handler() : { status: "error", message: "Action '" + action + "' belum terdaftar." });
  } catch (err) {
    return responseJSON({ status: "error", message: err.toString() });
  }
}

function responseJSON(data) {
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

// Operasi Spreadsheet Database
function getSheet(sheetName) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  let sheet = ss.getSheetByName(sheetName);
  if (!sheet) sheet = ss.insertSheet(sheetName);
  return sheet;
}

function sanitizeForSheet(value) {
  if (typeof value === "string" && /^[=+\-@]/.test(value)) return "'" + value;
  return value;
}

function getSheetData(sheetName) {
  const sheet = getSheet(sheetName);
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  const headers = data[0];
  const rows = [];
  for (let i = 1; i < data.length; i++) {
    let rowObj = {};
    for (let j = 0; j < headers.length; j++) {
      let val = data[i][j];
      if (headers[j] === "tanggal" || headers[j] === "created_at" || headers[j] === "jadwal_pantau") {
        val = formatDateWITA(val);
      }
      rowObj[headers[j]] = val;
    }
    rows.push(rowObj);
  }
  return rows;
}

function updateSheetRow(sheetName, id, updatedObj) {
  const sheet = getSheet(sheetName);
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return false;
  const headers = data[0];
  const idColIndex = headers.indexOf("id");

  for (let i = 1; i < data.length; i++) {
    if (String(data[i][idColIndex]) === String(id)) {
      for (let j = 0; j < headers.length; j++) {
        const key = headers[j];
        if (updatedObj[key] !== undefined) sheet.getRange(i + 1, j + 1).setValue(sanitizeForSheet(updatedObj[key]));
      }
      return true;
    }
  }
  return false;
}

function deleteRowById(sheetName, id) {
  const sheet = getSheet(sheetName);
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return { status: "error", message: "Data kosong." };

  const idColIndex = data[0].indexOf("id");
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][idColIndex]) === String(id)) {
      sheet.deleteRow(i + 1);
      return { status: "success", message: "Data berhasil dihapus." };
    }
  }
  return { status: "error", message: "ID tidak ditemukan." };
}

// Token Engine & Keamanan Password dengan CacheService
function getSecretKey() {
  const props = PropertiesService.getScriptProperties();
  let key = props.getProperty("APP_SECRET_KEY");
  if (!key) {
    key = Utilities.getUuid();
    props.setProperty("APP_SECRET_KEY", key);
  }
  return key;
}

function hashPassword(plainPassword) {
  const bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, String(plainPassword) + PASSWORD_PEPPER);
  return bytes.map(b => (b < 0 ? b + 256 : b).toString(16).padStart(2, '0')).join("");
}

function generateToken(userId, role) {
  const expiry = Date.now() + TOKEN_LIFETIME_MS;
  const payload = String(userId) + "|" + role + "|" + expiry;
  const encodedPayload = Utilities.base64EncodeWebSafe(payload);
  const signature = Utilities.base64EncodeWebSafe(Utilities.computeHmacSha256Signature(payload, getSecretKey()));
  return encodedPayload + "." + signature;
}

function validateToken(token) {
  if (!token || typeof token !== "string" || token.indexOf(".") === -1) return null;
  try {
    const cache = CacheService.getScriptCache();
    const cachedUser = cache.get("token_" + token);
    if (cachedUser) {
      return JSON.parse(cachedUser);
    }

    const parts = token.split(".");
    const payload = Utilities.newBlob(Utilities.base64DecodeWebSafe(parts[0])).getDataAsString();
    const expectedSig = Utilities.base64EncodeWebSafe(Utilities.computeHmacSha256Signature(payload, getSecretKey()));
    if (parts[1] !== expectedSig) return null;

    const p = payload.split("|");
    if (Number(p[2]) < Date.now()) return null;

    const sheetName = p[1] === "admin" ? "Admin" : (p[1] === "guru" ? "Guru" : "Siswa");
    const user = getSheetData(sheetName).find(r => String(r.id) === String(p[0]));
    if (!user) return null;

    delete user.password;
    user.role = p[1];

    // Simpan sesi valid di CacheService selama 15 menit (900 detik)
    cache.put("token_" + token, JSON.stringify(user), 900);
    return user;
  } catch (e) {
    return null;
  }
}

function handleRefreshToken(authUser) {
  const newToken = generateToken(authUser.id, authUser.role);
  logActivity(authUser.id, authUser.role, "REFRESH_TOKEN", { user_id: authUser.id });
  return { status: "success", token: newToken, user: authUser };
}

// Logika Otentikasi & Bootstrap
function handleLogin(payload) {
  const { role, username, password } = payload;
  if (!role || !username || !password) {
    return { status: "error", message: "Role, Username, dan Password wajib diisi." };
  }

  let sheetName = role === "admin" ? "Admin" : (role === "guru" ? "Guru" : "Siswa");
  const users = getSheetData(sheetName);
  const user = users.find(u => String(u.username).trim().toLowerCase() === String(username).trim().toLowerCase());

  if (!user) return { status: "error", message: "Kredensial pengguna tidak ditemukan." };

  const inputHash = hashPassword(password);
  const isMatch = (user.password === inputHash) || (String(user.password) === String(password));

  if (!isMatch) {
    return { status: "error", message: "Username atau Password salah." };
  }

  if (user.password !== inputHash) {
    updateSheetRow(sheetName, user.id, { password: inputHash });
  }

  const token = generateToken(user.id, role);
  delete user.password;
  user.role = role;
  
  logActivity(user.id, role, "LOGIN", { username: user.username });
  return { status: "success", token, user };
}

function handleGetBootstrapData(authUser) {
  const initial = handleGetInitialData(authUser);
  const dashboard = handleGetDashboardData(authUser);
  return {
    status: "success",
    data: {
      initial: initial.data,
      dashboard: dashboard.data
    }
  };
}

function handleGetInitialData(authUser) {
  const role = authUser.role;
  const kelasList = getSheetData("Kelas");
  const guruList = getSheetData("Guru").map(g => { delete g.password; return g; });
  const siswaList = getSheetData("Siswa").map(s => { delete s.password; return s; });

  let allowedSiswa = [];
  if (role === "admin") {
    allowedSiswa = siswaList;
  } else if (role === "guru") {
    const kls = kelasList.find(k => String(k.guru_id) === String(authUser.id));
    allowedSiswa = kls ? siswaList.filter(s => String(s.kelas_id) === String(kls.id)) : [];
  } else if (role === "siswa") {
    allowedSiswa = siswaList.filter(s => String(s.id) === String(authUser.id));
  }

  return {
    status: "success",
    data: {
      kelas: kelasList,
      guru: guruList,
      siswa: allowedSiswa
    }
  };
}

function handleGetDashboardData(authUser) {
  const role = authUser.role;
  const today = formatDateWITA(new Date());

  const allSiswa = getSheetData("Siswa").map(s => { delete s.password; return s; });
  const allKelas = getSheetData("Kelas");
  const allAbsensi = getSheetData("Kehadiran");
  const allAkademik = getSheetData("Akademik");
  const allPembinaan = getSheetData("Pembinaan");

  let mySiswaList = [];
  if (role === "admin") {
    mySiswaList = allSiswa;
  } else if (role === "guru") {
    const kls = allKelas.find(k => String(k.guru_id) === String(authUser.id));
    mySiswaList = kls ? allSiswa.filter(s => String(s.kelas_id) === String(kls.id)) : [];
  } else if (role === "siswa") {
    mySiswaList = allSiswa.filter(s => String(s.id) === String(authUser.id));
  }

  const mySiswaIds = mySiswaList.map(s => String(s.id));

  // Presensi hari ini
  const todayAbsensi = allAbsensi.filter(a => formatDateWITA(a.tanggal) === today && mySiswaIds.includes(String(a.siswa_id)));
  const totalHadirToday = todayAbsensi.filter(a => a.status === "H" || a.status === "T").length;

  // Indikator Perlu Perhatian
  const priorityList = [];
  const activePembinaanStatuses = ["Pemantauan", "Dalam pembinaan", "Perlu tindak lanjut"];

  mySiswaList.forEach(siswa => {
    const sId = String(siswa.id);
    const flags = [];

    const totalAlpa = allAbsensi.filter(a => String(a.siswa_id) === sId && a.status === "A").length;
    if (totalAlpa >= 3) {
      flags.push({ jenis: "absensi", pesan: `Banyak alpa (${totalAlpa} kali)` });
    }

    const lowGrades = allAkademik.filter(a => String(a.siswa_id) === sId && Number(a.nilai_akhir || 0) < Number(a.kktp || 75));
    if (lowGrades.length >= 2) {
      flags.push({ jenis: "akademik", pesan: `Nilai akademik kurang pada ${lowGrades.length} mapel` });
    }

    const activePem = allPembinaan.filter(p => String(p.siswa_id) === sId && activePembinaanStatuses.includes(String(p.status)));
    if (activePem.length > 0) {
      flags.push({ jenis: "pembinaan", pesan: `Ada ${activePem.length} catatan pembinaan aktif` });
    }

    if (flags.length > 0) {
      priorityList.push({
        siswa: siswa,
        indicators: flags
      });
    }
  });

  // Agenda Mendatang
  const agendaList = allPembinaan
    .filter(p => mySiswaIds.includes(String(p.siswa_id)) && p.jadwal_pantau && formatDateWITA(p.jadwal_pantau) >= today)
    .sort((a, b) => new Date(formatDateWITA(a.jadwal_pantau)) - new Date(formatDateWITA(b.jadwal_pantau)))
    .slice(0, 5)
    .map(p => {
      const s = mySiswaList.find(x => String(x.id) === String(p.siswa_id));
      return {
        id: p.id,
        siswa_id: p.siswa_id,
        nama_siswa: s ? s.nama : "Siswa",
        jadwal_pantau: formatDateWITA(p.jadwal_pantau),
        jenis: p.jenis,
        permasalahan: p.permasalahan,
        status: p.status
      };
    });

  return {
    status: "success",
    data: {
      total_siswa: mySiswaList.length,
      hadir_today: totalHadirToday,
      priority_list: priorityList,
      agenda_list: agendaList
    }
  };
}

function handleChangePassword(authUser, payload) {
  const { oldPassword, newPassword } = payload;
  if (!oldPassword || !newPassword) return { status: "error", message: "Password lama dan baru wajib diisi." };

  const sheetName = authUser.role === "admin" ? "Admin" : (authUser.role === "guru" ? "Guru" : "Siswa");
  const users = getSheetData(sheetName);
  const targetUser = users.find(u => String(u.id) === String(authUser.id));

  if (!targetUser || targetUser.password !== hashPassword(oldPassword)) {
    return { status: "error", message: "Password lama Anda tidak sesuai." };
  }

  updateSheetRow(sheetName, authUser.id, { password: hashPassword(newPassword) });
  logActivity(authUser.id, authUser.role, "CHANGE_PASSWORD", { user_id: authUser.id });
  return { status: "success", message: "Password berhasil diperbarui." };
}

function initDatabaseSheets() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const headersMap = {
    "Admin": ["id", "username", "password", "nama", "foto"],
    "Guru": ["id", "username", "password", "nama", "nip", "no_hp", "foto"],
    "Siswa": ["id", "username", "password", "nama", "nisn", "kelas_id", "no_hp_ortu", "foto"],
    "Kelas": ["id", "nama_kelas", "guru_id"],
    "Kehadiran": ["id", "siswa_id", "tanggal", "status", "waktu_masuk"],
    "Kebiasaan": ["id", "siswa_id", "tanggal", "kebiasaan_id", "status"],
    "Hafalan": ["id", "siswa_id", "tanggal", "nama_surat", "status", "catatan"],
    "Akademik": ["id", "siswa_id", "mapel", "nilai_akhir", "kktp"],
    "Prestasi": ["id", "siswa_id", "tanggal", "nama_prestasi", "tingkat", "kategori"],
    "Pembinaan": ["id", "siswa_id", "tanggal", "jenis", "permasalahan", "status", "jadwal_pantau"],
    "Logs": ["id", "timestamp", "user_id", "role", "action", "details"]
  };

  Object.keys(headersMap).forEach(sName => {
    let sheet = ss.getSheetByName(sName);
    if (!sheet) sheet = ss.insertSheet(sName);
    if (sheet.getLastRow() === 0) sheet.appendRow(headersMap[sName]);
  });

  const adminSheet = ss.getSheetByName("Admin");
  if (adminSheet.getDataRange().getValues().length <= 1) {
    const defaultAdminId = "ADM-001";
    const hashedPwd = hashPassword("admin123");
    adminSheet.appendRow([defaultAdminId, "admin", hashedPwd, "Administrator Utama", "https://ui-avatars.com/api/?name=Administrator"]);
    return { status: "success", message: "Database dan akun Admin default berhasil diinisialisasi." };
  }

  return { status: "success", message: "Struktur sheet dan database terverifikasi." };
}

// Master Guru, Siswa, Kelas Handlers
function handleSaveGuru(authUser, payload) {
  if (authUser.role !== "admin") return { status: "error", message: "Akses ditolak." };
  const { id, username, password, nama, nip, no_hp, foto } = payload;
  if (!username || !nama) return { status: "error", message: "Username dan Nama Guru wajib diisi." };

  const sheet = getSheet("Guru");
  const data = getSheetData("Guru");

  if (id) {
    const updatedObj = { username, nama, nip, no_hp, foto };
    if (password && password.trim() !== "") updatedObj.password = hashPassword(password);
    updateSheetRow("Guru", id, updatedObj);
    logActivity(authUser.id, authUser.role, "SAVE_GURU_UPDATE", { target_id: id });
    return { status: "success", message: "Data Guru berhasil diperbarui." };
  } else {
    const existing = data.find(g => String(g.username).toLowerCase() === String(username).toLowerCase());
    if (existing) return { status: "error", message: "Username sudah digunakan." };

    const newId = "GRU-" + Date.now();
    const finalPwd = password ? hashPassword(password) : hashPassword("guru123");
    sheet.appendRow([newId, username, finalPwd, nama, nip || "", no_hp || "", foto || ""]);
    logActivity(authUser.id, authUser.role, "SAVE_GURU_CREATE", { target_id: newId });
    return { status: "success", message: "Guru baru berhasil ditambahkan." };
  }
}

function handleDeleteGuru(authUser, payload) {
  if (authUser.role !== "admin") return { status: "error", message: "Akses ditolak." };
  logActivity(authUser.id, authUser.role, "DELETE_GURU", { target_id: payload.id });
  return deleteRowById("Guru", payload.id);
}

}

function handleSaveAbsensi(authUser, payload) {
  if (authUser.role === "siswa") return { status: "error", message: "Siswa tidak memiliki akses menyimpan presensi." };

  const tanggal = payload.tanggal || formatDateWITA(new Date());
  const items = payload.items || [];
  const sheet = getSheet("Kehadiran");
  const data = sheet.getDataRange().getValues();

  items.forEach(item => {
    const sId = String(item.siswa_id);
    const waktu = item.waktu_masuk || formatTimeWITA(new Date());
    let foundIndex = -1;

    for (let i = 1; i < data.length; i++) {
      if (String(data[i][1]) === sId && formatDateWITA(data[i][2]) === tanggal) {
        foundIndex = i + 1;
        break;
      }
    }

    if (foundIndex !== -1) {
      sheet.getRange(foundIndex, 4).setValue(sanitizeForSheet(item.status));
      sheet.getRange(foundIndex, 5).setValue(sanitizeForSheet(waktu));
    } else {
      const newId = "ABS-" + Date.now() + "-" + Math.floor(Math.random() * 1000);
      sheet.appendRow([newId, sId, tanggal, item.status, waktu]);
    }
  });

  logActivity(authUser.id, authUser.role, "SAVE_ABSENSI", { tanggal, count: items.length });
  return { status: "success", message: "Data presensi berhasil disimpan." };
}

// Detail Siswa & Laporan Rekap Handlers
function handleGetDetailSiswa(authUser, payload) {
  const siswaId = payload.siswa_id || authUser.id;
  const siswa = getSheetData("Siswa").find(s => String(s.id) === String(siswaId));
  if (!siswa) return { status: "error", message: "Data siswa tidak ditemukan." };

  const absensi = getSheetData("Kehadiran").filter(a => String(a.siswa_id) === String(siswaId));
  const kebiasaan = getSheetData("Kebiasaan").filter(k => String(k.siswa_id) === String(siswaId));
  const hafalan = getSheetData("Hafalan").filter(h => String(h.siswa_id) === String(siswaId));
  const akademik = getSheetData("Akademik").filter(ak => String(ak.siswa_id) === String(siswaId));
  const prestasi = getSheetData("Prestasi").filter(p => String(p.siswa_id) === String(siswaId));
  const pembinaan = getSheetData("Pembinaan").filter(pb => String(pb.siswa_id) === String(siswaId));

  return {
    status: "success",
    data: { siswa, absensi, kebiasaan, hafalan, akademik, prestasi, pembinaan }
  };
}

function handleGetLaporanRekap(authUser, payload) {
  const kelasId = payload.kelas_id;
  let allSiswa = getSheetData("Siswa");
  if (kelasId) {
    allSiswa = allSiswa.filter(s => String(s.kelas_id) === String(kelasId));
  }

  const absensi = getSheetData("Kehadiran");
  const akademik = getSheetData("Akademik");

  const rekap = allSiswa.map(s => {
    const sAbs = absensi.filter(a => String(a.siswa_id) === String(s.id));
    const hadir = sAbs.filter(a => a.status === 'H').length;
    const sakit = sAbs.filter(a => a.status === 'S').length;
    const izin = sAbs.filter(a => a.status === 'I').length;
    const alpa = sAbs.filter(a => a.status === 'A').length;

    const sAkd = akademik.filter(a => String(a.siswa_id) === String(s.id));
    const nilaiDibawahKKTP = sAkd.filter(a => Number(a.nilai_akhir) < Number(a.kktp)).length;

    return {
      id: s.id,
      nama: s.nama,
      nisn: s.nisn,
      kelas_id: s.kelas_id,
      presensi: { hadir, sakit, izin, alpa },
      total_mapel: sAkd.length,
      dibawah_kktp: nilaiDibawahKKTP
    };
  });

  return { status: "success", data: rekap };
}
